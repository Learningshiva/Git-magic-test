#Merge two sorted arrays into one sorted array using the array module.
#import array
alph = array.array('u',['u','a','c','b'])
alpha = array.array('u',['h','i','g','j'])

merge = array.array('u')
merge.extend(alph)
merge.extend(alpha)
merge = array.array('u', sorted(merge))
print("merge & sorted array: ", merge)

#Given a Unicode array array('u', 'pythonrocks'), replace 'o' with '0'.
unicode = array.array('u','pythonrocks')
for i in range(len(unicode)):
    if unicode[i] == 'o':
        unicode[i] = '0'
print(unicode)
